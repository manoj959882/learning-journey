    public static ListListInteger mergeOverlap(ListListInteger intervals) {
        ListListInteger ans=new ArrayList();
        if (intervals.isEmpty()) return ans;
        intervals.sort((a,b)-a.get(1)-b.get(0));
        for (ListInteger elemIntervalintervals){
            if (ans.isEmpty() ans.get(ans.size()-1).get(1)elemInterval.get(0)){
                ans.add(elemInterval);
            }
            else{
                int newEnd=Math.max(elemInterval.get(1),ans.get(ans.size()-1).get(1));
                ans.get(ans.size()-1).set(1,newEnd);
            }
        }
        return ans;
    }

    public static int[][] merge(int[][] intervals) {
        if (intervals == null  intervals.length = 1) {
            return intervals;
        }
        Arrays.sort(intervals,(a,b)-a[0]-b[0]);
        Listint[] result = new ArrayList();
        int[] current = intervals[0];
        result.add(current);

        for (int[] interval  intervals) {
            int currentEnd=current[1];
            int nextStart=interval[0];
            int nextEnd=interval[1];
            if (nextStart=currentEnd){
                current[1]=Math.max(currentEnd,nextEnd);
            }
            else{
                current=interval;
                result.add(current);
            }
        }
        return result.toArray(new int[result.size()][]);

    }

    public static long  mergeSort(int[] nums,int left,int right){
        if (left==right) return 0;
        int finalCount=0;
        if (leftright) {
            int mid=left+(right-left)2;
           long leftCount= mergeSort(nums,left,mid);
           long rightCount= mergeSort(nums,mid+1,right);
           long mergeCount = merge(nums,left,mid,right);
           finalCount+= leftCount+rightCount+mergeCount;
        }
        return finalCount;
    }

    public static void mergeTwoSortedArray(int[] nums1, int m, int[] nums2, int n) {
        int i=m-1,j=n-1,k=m+n-1;
        while (i = 0 && j = 0){
            if (nums1[i]  nums2[j]){
                nums1[k] = nums1[i];
                i--; k--;
            }
            if (nums1[i] = nums2[j]){
                nums1[k] = nums2[j] ;
                j--; k--;
            }
        }
        while (j = 0) {
            nums1[k] = nums2[j];
            j--;
            k--;
        }
    }

    public static long merge(int[] nums,int left,int mid,int right){
        int i=left,j=right;
        int k=0;
        int count=0;
        int[] temp=new int[right-left+1];
        while (i=mid&&j=right){
            if (nums[i]=nums[j]){
                temp[k++] = nums[i++];
            }

            else if (nums[i]nums[j]){
                count+=(mid-i+1);
                temp[k++] = nums[j++];

            }
        }
        while (i = mid){
            temp[k] = nums[i];
            i++;
            k++;
        }
        while (j=right){
            temp[k++]=nums[j++];

        }
        for (int p=0;ptemp.length;p++){
            nums[left + p] = temp[p];
        }
        return count;

    }

    public static int maxProduct(int[] nums) {
        if (nums == null  nums.length == 0) {
            return 0;}
        int n = nums.length;
        int maxProduct = nums[0];
        int minProduct = nums[0];
        int answer = nums[0];
        for (int i = 1; i  n; i++) {
            if (nums[i]  0) {
                int temp = maxProduct;
                maxProduct = minProduct;
                minProduct = temp;
            }
             Update max and min products ending at the current position
            maxProduct = Math.max(nums[i], maxProduct  nums[i]);
            minProduct = Math.min(nums[i], minProduct  nums[i]);
             Update the global result
            answer = Math.max(answer, maxProduct);
        }
        return answer;
    }

    public static int[] productExceptSelf(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];

        int leftProduct = 1;
        for (int i = 0; i  n; i++) {
            result[i] = leftProduct;       Store product of all elements to the left
            leftProduct = nums[i];         Update left product for next iteration
        }

         Second pass Calculate products of all elements to the right of each index
         Multiply each position by the product of all elements after index i
        int rightProduct = 1;
        for (int i = n - 1; i = 0; i--) {
            result[i] = rightProduct;      Multiply by product of all elements to the right
            rightProduct = nums[i];        Update right product for next iteration
        }

        return result;

    }

    public static int searchInsert(int[] nums, int target) {
        int left =0,right=nums.length-1;
        while (left=right){
            int mid=left+(right-left)2;
            if (nums[mid]==target){
                return mid;
            }
            if (nums[mid]target){
                left=mid+1;
            }
            else {
                right=mid-1;
            }
        }
        return left;

    }

    public static int numSubarrayProductLessThanK(int[] nums, int k) {
        if (k = 1) return 0;   Important edge case

        int left = 0;
        int product = 1;
        int count = 0;

        for (int right = 0; right  nums.length; right++) {
            product = nums[right];

             Jab product = k ho, tab left pointer ko aage badhao
            while (product = k && left = right) {
                product = nums[left];
                left++;
            }

             Current window ke andar kitne subarrays valid hain
            count += right - left + 1;
        }

        return count;

    }

    public static int findMin(int[] nums) {
        int left=0,right=nums.length-1;
        while (leftright){
            int mid=left+(right-left)2;
            if (nums[mid]=nums[right]){
                left=mid+1;
            }
            else {
                right=mid;
            }
        }
        return nums[left];
    }

    public static int singleNonDuplicate(int[] nums) {
        int left = 0, right = nums.length - 1;
        while (left  right) {
            int mid = left + (right - left)  2;
             Step 1 mid ko hamesha EVEN banao
             Kyunki humari saari logic even index ke pattern pe based hai
            if (mid % 2 == 1) {
                mid--;   odd hai to ek peeche kar do, ab even ho jaayega
            }
             Step 2 check karo pattern normal hai ya shift ho gaya
            if (nums[mid] == nums[mid + 1]) {
                 Pattern normal hai - single element right side mein hai
                 mid aur mid+1 dono discard, kyunki ye pura pair hai
                left = mid + 2;
            } else {
                 Pattern shift ho gaya - single element mid ya left mein hai
                 mid ko discard nahi karte, kyunki khud answer ho sakta hai
                right = mid;
            }
        }
         Jab left == right, wahi single element hai
        return nums[left];
    }

    public static int longestSumWithZero(int[] a) {
        int maxLen = 0;
        MapInteger, Integer hmap = new HashMap();
        int sum = 0;
        for (int i = 0; i  a.length; i++) {
            sum += a[i];
            if (sum == 0) {
                maxLen = i + 1;
            }
            else if (hmap.containsKey(sum)) {
                maxLen = Math.max(maxLen, i - hmap.get(sum));
            }
            else {
                hmap.put(sum, i);
            }
        }
        return maxLen;
    }

    public static int longestSubarraySumK(int[] arr, int k){
        MapInteger, Integer hmap = new HashMap();
        int sum = 0;
        int maxLen = 0;
        for (int i = 0; i  arr.length; i++) {
            sum += arr[i];
            if (sum == k) {
                maxLen = i + 1;
            }
            if (hmap.containsKey(sum-k)) {
                maxLen = Math.max(maxLen, i - hmap.get(sum-k));
            }
            if (!hmap.containsKey(sum)) {
                hmap.put(sum, i);
            }
        }
        return maxLen;
    }

    public static ListInteger getRow(int rowIndex) {
        ListInteger row = new ArrayList();
         Row 0 hamesha [1] se start hoti hai, isliye pehla element 1 daal diya
        row.add(1);
         i loop chalega 1 se rowIndex tak, kyunki row 0 already ban gayi hai upar
        for (int i = 1; i = rowIndex; i++) {
             Har nayi row banane se pehle, list ke end mein 1 add kar do
             kyunki har row ka last element hamesha 1 hota hai (Pascal's triangle property)
            row.add(1);
             j loop right se left chalega (i-1 se 1 tak)
             right se left chalne ka reason agar left se chalte, toh row.get(j-1)
             already update ho chuki hoti current row ki value se, jo galat hoga
            for (int j = i - 1; j  0; j--) {
                 Current position ki value = (upar wali row ka current element) + (upar wali row ka pehle wala element)
                 row.get(j) abhi bhi purani (upar wali) row ki value hai kyunki hum right se left ja rahe hain
                 row.get(j-1) bhi abhi purani row ki value hai (update nahi hui abhi tak)
                row.set(j, row.get(j) + row.get(j - 1));
            }
             j=0 wala element kabhi change nahi hota, wo hamesha 1 rehta hai
             (Pascal's triangle mein har row ka first element 1 hota hai), isliye loop j0 tak hi chalaya
        }
        return row;

    }



    public static ListInteger spiralOrder(int[][] matrix) {
        int top=0,left=0,right=matrix[0].length-1,bottom=matrix.length-1;
        ListInteger ans=new ArrayList();
        while (top=bottom&&left=right){setting boundary to execute our code.
            for (int i=left;i=right;i++){
                ans.add(matrix[top][i]);top liya hun Qki top will not be changed as row will be same here
            }
            top++;increase bcoz we will now go for second row.
            for(int i=top;i=bottom;i++){top se bottom
                ans.add(matrix[i][right]);
            }
            right--;decrease QKi 9 tak cover kar gaye hai ab 8 se number suru hoga.
            if(top=bottom){right se left ki taraf
                for(int i=right;i=left;i--){
                    ans.add(matrix[bottom][i]);
                }
            }
            bottom--;
            if(left=right){bottom se top ki taraf
                for(int i=bottom;i=top;i--){
                    ans.add(matrix[i][left]);
                }
            }
            left++;
        }
        return ans;

    }

    public static ListListInteger pascalTriangle(int numRows) {
        ListListInteger ans=new ArrayList();
        for (int i=0;i=numRows;i++){
            ListInteger currentRows=new ArrayList();
            for (int j=0;j=i;j++){
                if (j==0j==i){
                    currentRows.add(1);
                }
                else {
                    ListInteger prevRow  =new ArrayList(ans.get(i-1));
                    currentRows.add(prevRow.get(j-1)+prevRow.get(j));
                }
            }
            ans.add(currentRows);
        }
        return ans;
    }

    public static int[][] setZeroes(int[][] matrix) {
        boolean[] rowMark=new boolean[matrix.length];
        boolean[] colMark=new boolean[matrix[0].length];
        for (int i=0;imatrix.length;i++){
            for (int j=0;j matrix[0].length;j++){
                if (matrix[i][j]==0){
                    rowMark[i]=true;
                    colMark[j]=true;
                }
            }
        }

        for (int i=0;imatrix.length;i++){
            if (rowMark[i]){
                for (int j=0;jmatrix[0].length;j++){
                    matrix[i][j]=0;
                }
            }
        }

        for (int i=0;imatrix[0].length;i++){
            if (colMark[i]){
                for (int j=0;jmatrix.length;j++){
                    matrix[j][i]=0;
                }
            }
        }

        return matrix;
    }

    public static void nextPermutation(int[] nums) {
        int breakPoint=-1;
        int n=nums.length;
        for (int i=n-2;i=0;i--){
            if (nums[i]nums[i+1]){
                breakPoint=i;
                break;
            }
        }
        if (breakPoint==-1) {
            reverse(nums,0,n-1);
        }
        for (int i=n-1;ibreakPoint;i--){
            if (nums[i]nums[breakPoint]){
                swap(nums,i,breakPoint);
                break;
            }
        }
    }

    public static void reverse(int [] arr,int i,int j){
        while (ij){
            swap(arr,i++,j--);
        }
    }

    public static void swap(int[] arr,int i,int j){
        int temp=arr[i];
        arr[i]=arr[j];
        arr[j]=temp;
    }
